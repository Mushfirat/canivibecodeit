# --- canivibecodeit.com — append to /etc/caddy/Caddyfile once Cloudflare zones exist ---

canivibecodeit.com {
	tls {
		dns cloudflare {env.CLOUDFLARE_API_TOKEN}
	}
	import posthog
	encode gzip zstd

	# static assets are content-hashed by Astro
	@assets path /_astro/* /icons/* /og/* /js/* /fonts/*
	header @assets Cache-Control "public, max-age=31536000, immutable"

	header {
		X-Content-Type-Options nosniff
		Referrer-Policy strict-origin-when-cross-origin
	}

	reverse_proxy 127.0.0.1:8095

	log {
		output file /srv/http/canivibecodeit.com/logs/caddy-access.log
	}
}

# typo/brand cover — 301 everything to the primary, preserving path + query
canivibecode.com, www.canivibecode.com,

canivibecode.it, www.canivibecode.it,
www.canivibecodeit.com {
	tls {
		dns cloudflare {env.CLOUDFLARE_API_TOKEN}
	}
	redir https://canivibecodeit.com{uri} permanent
}
